cmake_minimum_required(VERSION 3.16)
project(quant VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

option(QUANT_BUILD_DAEMON "Build quantd / quant-cli" ON)
option(QUANT_BUILD_TESTS  "Build tests" ON)
option(QUANT_UPNP         "Build with UPnP (miniupnpc)" ON)

# ---------------------------------------------------------------- BLAKE3
set(B3 ${CMAKE_CURRENT_SOURCE_DIR}/third_party/blake3)
set(BLAKE3_SRC ${B3}/blake3.c ${B3}/blake3_dispatch.c ${B3}/blake3_portable.c)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
  list(APPEND BLAKE3_SRC ${B3}/blake3_sse2.c ${B3}/blake3_sse41.c ${B3}/blake3_avx2.c ${B3}/blake3_avx512.c)
  set_source_files_properties(${B3}/blake3_sse2.c   PROPERTIES COMPILE_OPTIONS "-msse2")
  set_source_files_properties(${B3}/blake3_sse41.c  PROPERTIES COMPILE_OPTIONS "-msse4.1")
  set_source_files_properties(${B3}/blake3_avx2.c   PROPERTIES COMPILE_OPTIONS "-mavx2")
  set_source_files_properties(${B3}/blake3_avx512.c PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512vl")
  set(BLAKE3_DEFS "")
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64|ARM64")
  list(APPEND BLAKE3_SRC ${B3}/blake3_neon.c)
  set(BLAKE3_DEFS BLAKE3_USE_NEON=1)
else()
  set(BLAKE3_DEFS BLAKE3_NO_SSE2 BLAKE3_NO_SSE41 BLAKE3_NO_AVX2 BLAKE3_NO_AVX512)
endif()
add_library(blake3 STATIC ${BLAKE3_SRC})
target_include_directories(blake3 PUBLIC ${B3})
target_compile_definitions(blake3 PRIVATE ${BLAKE3_DEFS})

# ---------------------------------------------------------------- PQClean (Falcon-512 + SPHINCS+)
set(PQ ${CMAKE_CURRENT_SOURCE_DIR}/third_party/pqclean)
add_library(pqcommon STATIC ${PQ}/common/fips202.c ${PQ}/common/sha2.c)
target_include_directories(pqcommon PUBLIC ${PQ}/common)
file(GLOB FALCON_SRC ${PQ}/falcon512/*.c)
add_library(falcon512 STATIC ${FALCON_SRC})
target_include_directories(falcon512 PRIVATE ${PQ}/falcon512)
target_link_libraries(falcon512 PUBLIC pqcommon)
file(GLOB SPHINCS_SRC ${PQ}/sphincs/*.c)
add_library(sphincs STATIC ${SPHINCS_SRC})
target_include_directories(sphincs PRIVATE ${PQ}/sphincs)
target_link_libraries(sphincs PUBLIC pqcommon)

# ---------------------------------------------------------------- miniupnpc
if(QUANT_UPNP)
  set(UPNPC_BUILD_STATIC ON CACHE BOOL "" FORCE)
  set(UPNPC_BUILD_SHARED OFF CACHE BOOL "" FORCE)
  set(UPNPC_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(UPNPC_BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
  set(UPNPC_NO_INSTALL ON CACHE BOOL "" FORCE)
  add_subdirectory(third_party/miniupnpc EXCLUDE_FROM_ALL)
endif()

# ---------------------------------------------------------------- libquant
file(GLOB_RECURSE QUANT_SRC CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
list(FILTER QUANT_SRC EXCLUDE REGEX "/(quantd|quant-cli)\\.cpp$")
add_library(quant STATIC ${QUANT_SRC})
target_include_directories(quant PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src ${CMAKE_CURRENT_SOURCE_DIR}/third_party/json)
target_include_directories(quant PRIVATE ${PQ}/falcon512 ${PQ}/sphincs)
target_link_libraries(quant PUBLIC blake3 falcon512 sphincs pqcommon)
find_package(Threads REQUIRED)
target_link_libraries(quant PUBLIC Threads::Threads)
if(QUANT_UPNP)
  target_link_libraries(quant PUBLIC libminiupnpc-static)
  target_compile_definitions(quant PUBLIC QUANT_UPNP=1)
endif()
if(WIN32)
  target_link_libraries(quant PUBLIC ws2_32 iphlpapi bcrypt)
  target_compile_definitions(quant PUBLIC _WIN32_WINNT=0x0601 NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
target_compile_options(quant PRIVATE -Wall -Wextra -Wno-unused-parameter)

if(QUANT_BUILD_DAEMON)
  add_executable(quantd src/quantd.cpp)
  target_link_libraries(quantd PRIVATE quant)
  add_executable(quant-cli src/quant-cli.cpp)
  target_link_libraries(quant-cli PRIVATE quant)
  if(WIN32)
    target_link_options(quantd PRIVATE -static)
    target_link_options(quant-cli PRIVATE -static)
  endif()
endif()

if(QUANT_BUILD_TESTS)
  add_executable(quant-tests tests/test_main.cpp)
  target_link_libraries(quant-tests PRIVATE quant)
  if(WIN32)
    target_link_options(quant-tests PRIVATE -static)
  endif()
endif()
